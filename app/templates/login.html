from openai import OpenAI

client = OpenAI(
    base_url="{{ api_endpoint }}",
    api_key="YOUR_API_KEY",
)

response = client.chat.completions.create(
    model="{{ free_models[0] }}",
    messages=[{
        "role": "user",
        "content": "Let's build something great."
    }],
)

print(response.choices[0].message.content)
